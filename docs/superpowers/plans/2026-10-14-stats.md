**Goal:** Экран «Статистика» в вебе и боте: деньги, поток машин, сотрудники, работы и клиенты для админа; «моё» для мастера и механика.

**Spec:** `docs/superpowers/specs/2026-10-14-stats-design.md`. Макет: https://claude.ai/artifact/6wH4dSywoLYGnB3AEtyeup

## Global Constraints

- Ветка `stats` (ответвлена от `admin-assignee`); в `main` вливаются вместе по команде владельца.
- Зарплаты вне объёма.
- Доступ: `/stats/money|flow|staff|works` — только админ; `/stats/mine` — только мастер и механик; остальным 403.
- `master_id` — только у админских разделов: сузить до заездов одного мастера.
- Периоды скользящие по Москве (`Europe/Moscow`): week 7 дн. (шаг день), month 30 (шаг неделя), quarter 90 (шаг неделя), year 365 (шаг календарный месяц); прошлое окно той же длины перед текущим.
- `delta_pct = round((cur - prev) / prev * 100)`, при `prev = 0` — `null`.
- Механику — ни одного KPI или графика с `unit = "rub"`.
- Без новых таблиц и миграций; без новых npm/pip зависимостей.
- Палитра графиков: серия 1 `#3987e5`, серия 2 `#d95926`; янтарь — только акцент интерфейса.
- Тексты интерфейса — на русском; нормо-часы в вебе пишутся «н/ч», как в остальном приложении.
- Перед тестами: `docker compose up -d postgres` (после рестарта WSL контейнер падает).
- Команды: бэкенд `uv run pytest -q`, веб `cd web && npx vitest run && npx tsc --noEmit -p .`.

```python
# app/modules/stats/periods.py
"""Rolling stats windows in Moscow time, their chart steps, and period-over-period change."""
from dataclasses import dataclass
from datetime import date, datetime, time, timedelta
from typing import Literal
from zoneinfo import ZoneInfo

MSK = ZoneInfo("Europe/Moscow")
Period = Literal["week", "month", "quarter", "year"]
PERIOD_DAYS: dict[str, int] = {"week": 7, "month": 30, "quarter": 90, "year": 365}
MONTHS = ["янв", "фев", "мар", "апр", "май", "июн", "июл", "авг", "сен", "окт", "ноя", "дек"]


@dataclass(frozen=True)
class Window:
    start: datetime  # inclusive
    end: datetime  # exclusive

    def __contains__(self, moment: datetime) -> bool:
        return self.start <= moment < self.end


def _midnight(day: date) -> datetime:
    return datetime.combine(day, time(), tzinfo=MSK)


def windows(period: Period, now: datetime | None = None) -> tuple[Window, Window]:
    """The window ending with today (Moscow) and the same-length window right before it."""
    today = (now or datetime.now(MSK)).astimezone(MSK).date()
    end = _midnight(today + timedelta(days=1))
    length = timedelta(days=PERIOD_DAYS[period])
    current = Window(end - length, end)
    return current, Window(current.start - length, current.start)


def buckets(period: Period, window: Window) -> list[tuple[str, Window]]:
    """Chart steps covering `window`: days (week), calendar months (year), else 7-day steps ending today."""
    steps: list[tuple[str, Window]] = []
    if period == "week":
        start = window.start
        while start < window.end:
            end = start + timedelta(days=1)
            steps.append((f"{start:%d.%m}", Window(start, end)))
            start = end
        return steps
    if period == "year":
        start = window.start
        while start < window.end:
            following = _midnight(date(start.year + start.month // 12, start.month % 12 + 1, 1))
            end = min(following, window.end)
            steps.append((f"{MONTHS[start.month - 1]} {start:%y}", Window(start, end)))
            start = end
        return steps
    end = window.end
    while end > window.start:
        start = max(end - timedelta(days=7), window.start)
        steps.append((f"{start:%d.%m}–{end - timedelta(days=1):%d.%m}", Window(start, end)))
        end = start
    return steps[::-1]


def delta_pct(current: float, previous: float) -> int | None:
    if not previous:
        return None
    return round((current - previous) / previous * 100)
```

```python
# app/modules/stats/schemas.py
"""One shape for every stats section: KPI tiles plus charts. Labels are Russian so web and bot read the same."""
from typing import Literal

from pydantic import BaseModel

from app.modules.stats.periods import Period

Unit = Literal["rub", "count", "hours", "days", "pct", "text"]
ChartKind = Literal["bars", "stack", "hbars", "share"]


class Kpi(BaseModel):
    key: str
    label: str
    value: float | str
    unit: Unit
    delta_pct: int | None = None
    compare: bool = True  # False: a current-state figure with no previous period to compare to


class Series(BaseModel):
    name: str
    values: list[float]


class Chart(BaseModel):
    key: str
    kind: ChartKind
    title: str
    sub: str | None = None
    unit: Unit
    labels: list[str]
    series: list[Series]


class StatsOut(BaseModel):
    section: str
    period: Period
    kpis: list[Kpi]
    charts: list[Chart]
```

```python
def _alive(stmt: Select) -> Select:
    """Only visits whose visit, client and vehicle are not deleted."""
    return (
        stmt.join(Client, Client.id == Visit.client_id)
        .join(Vehicle, Vehicle.id == Visit.vehicle_id)
        .where(Visit.deleted_at.is_(None), Client.deleted_at.is_(None), Vehicle.deleted_at.is_(None))
    )


class StatsRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def _rows(self, stmt) -> list[Row]:
        return list((await self.session.execute(stmt)).all())

    async def closed_visits(self, status: VisitStatus, window: Window, master_id: uuid.UUID | None = None) -> list[Row]:
        """Visits now in `status` (issued/cancelled) that entered it inside `window`.
        A visit without that log entry (old or imported data) counts as closed when created."""
        last = (
            select(VisitStatusLog.visit_id, func.max(VisitStatusLog.changed_at).label("at"))
            .where(VisitStatusLog.to_status == status)
            .group_by(VisitStatusLog.visit_id)
            .subquery()
        )
        closed_at = func.coalesce(last.c.at, Visit.created_at)
        stmt = _alive(
            select(
                Visit.id, Visit.created_at, Visit.total_amount, Visit.client_id,
                Visit.assigned_master_id.label("master_id"), User.full_name.label("master_name"),
                closed_at.label("closed_at"),
            )
            .select_from(Visit)
            .outerjoin(last, last.c.visit_id == Visit.id)
            .join(User, User.id == Visit.assigned_master_id)
        ).where(Visit.status == status, closed_at >= window.start, closed_at < window.end)
        if master_id is not None:
            stmt = stmt.where(Visit.assigned_master_id == master_id)
        return await self._rows(stmt.order_by(closed_at, Visit.id))

    async def created_visits(self, window: Window, master_id: uuid.UUID | None = None) -> list[Row]:
        stmt = _alive(select(Visit.id, Visit.created_at, Visit.client_id, Vehicle.make).select_from(Visit)).where(
            Visit.created_at >= window.start, Visit.created_at < window.end
        )
        if master_id is not None:
            stmt = stmt.where(Visit.assigned_master_id == master_id)
        return await self._rows(stmt.order_by(Visit.created_at, Visit.id))

    async def open_count(self, master_id: uuid.UUID | None = None) -> int:
        stmt = _alive(select(func.count(Visit.id)).select_from(Visit)).where(Visit.status.not_in(CLOSED_STATUSES))
        if master_id is not None:
            stmt = stmt.where(Visit.assigned_master_id == master_id)
        return (await self.session.execute(stmt)).scalar_one()

    async def work_lines(self, visit_ids: list[uuid.UUID]) -> list[Row]:
        if not visit_ids:
            return []
        mechanic = aliased(User)
        stmt = (
            select(
                VisitWorkItem.visit_id,
                func.coalesce(WorkCatalog.name, VisitWorkItem.free_text_name).label("name"),
                VisitWorkItem.category, VisitWorkItem.norm_hours, VisitWorkItem.hourly_rate, VisitWorkItem.status,
                VisitWorkItem.assigned_mechanic_id.label("mechanic_id"),
                mechanic.full_name.label("mechanic_name"), mechanic.role.label("mechanic_role"),
                VisitWorkItem.is_extra_work, VisitWorkItem.approved_by_client,
            )
            .outerjoin(WorkCatalog, WorkCatalog.id == VisitWorkItem.catalog_item_id)
            .outerjoin(mechanic, mechanic.id == VisitWorkItem.assigned_mechanic_id)
            .where(VisitWorkItem.visit_id.in_(visit_ids))
            .order_by(VisitWorkItem.created_at, VisitWorkItem.id)
        )
        return await self._rows(stmt)

    async def part_sums(self, visit_ids: list[uuid.UUID]) -> dict[uuid.UUID, float]:
        if not visit_ids:
            return {}
        stmt = (
            select(VisitPartItem.visit_id, func.sum(VisitPartItem.quantity * VisitPartItem.unit_price))
            .where(VisitPartItem.visit_id.in_(visit_ids))
            .group_by(VisitPartItem.visit_id)
        )
        return {visit_id: float(total) for visit_id, total in await self._rows(stmt)}

    async def status_logs(self, visit_ids: list[uuid.UUID]) -> list[Row]:
        if not visit_ids:
            return []
        stmt = (
            select(VisitStatusLog.visit_id, VisitStatusLog.to_status, VisitStatusLog.changed_at)
            .where(VisitStatusLog.visit_id.in_(visit_ids))
            .order_by(VisitStatusLog.changed_at)
        )
        return await self._rows(stmt)

    async def first_visit_at(self, client_ids: list[uuid.UUID]) -> dict[uuid.UUID, datetime]:
        if not client_ids:
            return {}
        stmt = (
            _alive(select(Visit.client_id, func.min(Visit.created_at)).select_from(Visit))
            .where(Visit.client_id.in_(client_ids))
            .group_by(Visit.client_id)
        )
        return dict(await self._rows(stmt))

    async def open_work_count(self, mechanic_id: uuid.UUID) -> int:
        """Not-yet-ready works assigned to `mechanic_id` in open visits."""
        stmt = _alive(
            select(func.count(VisitWorkItem.id)).select_from(VisitWorkItem).join(Visit, Visit.id == VisitWorkItem.visit_id)
        ).where(
            VisitWorkItem.assigned_mechanic_id == mechanic_id,
            VisitWorkItem.status != WorkItemStatus.READY,
            Visit.status.not_in(CLOSED_STATUSES),
        )
        return (await self.session.execute(stmt)).scalar_one()
```

```python
"""Stats sections. Each returns StatsOut: KPI tiles (with change against the previous
window) and charts whose labels are the window's steps or ranked rows."""
import uuid
from collections import defaultdict
from datetime import datetime

from sqlalchemy.ext.asyncio import AsyncSession

from app.core.enums import UserRole, VisitStatus, WorkCategory
from app.modules.stats.periods import Period, Window, buckets, delta_pct, windows
from app.modules.stats.repository import StatsRepository
from app.modules.stats.schemas import Chart, ChartKind, Kpi, Series, StatsOut, Unit

FUNNEL = [
    VisitStatus.RECEIVED, VisitStatus.DIAGNOSTICS, VisitStatus.APPROVAL,
    VisitStatus.IN_PROGRESS, VisitStatus.WAITING_PARTS, VisitStatus.READY,
]
STATUS_LABELS = {
    VisitStatus.RECEIVED: "Принят", VisitStatus.DIAGNOSTICS: "Диагностика", VisitStatus.APPROVAL: "Согласование",
    VisitStatus.IN_PROGRESS: "Ремонт", VisitStatus.WAITING_PARTS: "Ждём запчасти", VisitStatus.READY: "Готов",
}
CATEGORY_LABELS = {
    WorkCategory.DIAGNOSTICS: "Диагностика", WorkCategory.MAINTENANCE: "ТО", WorkCategory.BODY: "Кузовные",
    WorkCategory.ELECTRICAL: "Электрика", WorkCategory.CHASSIS: "Ходовая", WorkCategory.OTHER: "Прочее",
}
DAY_SECONDS = 86400

Steps = list[tuple[str, Window]]


def _bucketed(steps: Steps, points) -> list[float]:
    """Sum (moment, amount) points into the steps they fall in."""
    values = [0.0] * len(steps)
    for moment, amount in points:
        for i, (_, window) in enumerate(steps):
            if moment in window:
                values[i] += float(amount)
                break
    return [round(v, 2) for v in values]


def _kpi(key: str, label: str, value: float | str, unit: Unit, previous: float | None = None, compare: bool = True) -> Kpi:
    delta = delta_pct(value, previous) if compare and previous is not None else None
    return Kpi(key=key, label=label, value=value, unit=unit, delta_pct=delta, compare=compare)


def _chart(key: str, kind: ChartKind, title: str, unit: Unit, labels: list[str], *series: tuple[str, list[float]],
           sub: str | None = None) -> Chart:
    return Chart(key=key, kind=kind, title=title, sub=sub, unit=unit, labels=labels,
                 series=[Series(name=name, values=values) for name, values in series])


def _ranked(totals: dict[str, float]) -> tuple[list[str], list[float]]:
    """Rows sorted by value, largest first; ties by label so the order is stable."""
    rows = sorted(totals.items(), key=lambda item: (-item[1], item[0]))
    return [label for label, _ in rows], [round(value, 2) for _, value in rows]


def _labor(line) -> float:
    return float(line.norm_hours) * float(line.hourly_rate)


class StatsService:
    def __init__(self, session: AsyncSession, now: datetime | None = None):
        self.repo = StatsRepository(session)
        self.now = now

    def _windows(self, period: Period) -> tuple[Window, Window, Steps]:
        current, previous = windows(period, self.now)
        return current, previous, buckets(period, current)

    async def money(self, period: Period, master_id: uuid.UUID | None = None) -> StatsOut:
        current, previous, steps = self._windows(period)
        issued = await self.repo.closed_visits(VisitStatus.ISSUED, current, master_id)
        issued_prev = await self.repo.closed_visits(VisitStatus.ISSUED, previous, master_id)
        revenue = sum(float(v.total_amount) for v in issued)
        revenue_prev = sum(float(v.total_amount) for v in issued_prev)
        average = revenue / len(issued) if issued else 0
        average_prev = revenue_prev / len(issued_prev) if issued_prev else 0

        ids = [v.id for v in issued]
        labor: dict[uuid.UUID, float] = defaultdict(float)
        for line in await self.repo.work_lines(ids):
            labor[line.visit_id] += _labor(line)
        parts = await self.repo.part_sums(ids)
        labor_total, parts_total = sum(labor.values()), sum(parts.values())
        if labor_total + parts_total:
            share = round(labor_total / (labor_total + parts_total) * 100)
            split = f"{share} / {100 - share} %"
        else:
            split = "—"

        labels = [label for label, _ in steps]
        return StatsOut(
            section="money", period=period,
            kpis=[
                _kpi("revenue", "Выручка", round(revenue, 2), "rub", revenue_prev),
                _kpi("avg_check", "Средний чек", round(average), "rub", average_prev),
                _kpi("issued", "Выдано заездов", len(issued), "count", len(issued_prev)),
                _kpi("work_share", "Работы / запчасти", split, "text", compare=False),
            ],
            charts=[
                _chart("revenue", "bars", "Выручка", "rub", labels,
                       ("Выручка", _bucketed(steps, ((v.closed_at, v.total_amount) for v in issued))),
                       sub="по дате выдачи машины"),
                _chart("revenue_split", "stack", "Из чего выручка", "rub", labels,
                       ("Работы", _bucketed(steps, ((v.closed_at, labor[v.id]) for v in issued))),
                       ("Запчасти", _bucketed(steps, ((v.closed_at, parts.get(v.id, 0)) for v in issued))),
                       sub="до скидки"),
            ],
        )
```

```python
def _days(seconds: float) -> float:
    return round(seconds / DAY_SECONDS, 1)


def _time_in_statuses(visits, logs) -> tuple[list[float], float]:
    """Average days a closed visit spent in each funnel status, and from intake to close."""
    if not visits:
        return [0.0] * len(FUNNEL), 0.0
    history = defaultdict(list)
    for row in logs:
        history[row.visit_id].append((row.to_status, row.changed_at))
    spent: dict[VisitStatus, float] = defaultdict(float)
    through = 0.0
    for v in visits:
        steps = [(VisitStatus.RECEIVED, v.created_at), *sorted(history[v.id], key=lambda step: step[1])]
        for (status, start), (_, end) in zip(steps, steps[1:]):
            if status in STATUS_LABELS:
                spent[status] += (end - start).total_seconds()
        through += (v.closed_at - v.created_at).total_seconds()
    return [_days(spent[s] / len(visits)) for s in FUNNEL], _days(through / len(visits))
```

```python
    async def works(self, period: Period, master_id: uuid.UUID | None = None) -> StatsOut:
        current, previous, _ = self._windows(period)
        issued = await self.repo.closed_visits(VisitStatus.ISSUED, current, master_id)
        lines = await self.repo.work_lines([v.id for v in issued])

        new, repeat = await self._clients(created, current)
        new_prev, repeat_prev = await self._clients(created_prev, previous)
        approved = await self._extra_approved(created)
        approved_prev = await self._extra_approved(created_prev)

        counts: dict[str, float] = defaultdict(float)
        by_category: dict[str, float] = defaultdict(float)
        for line in lines:

            by_category[CATEGORY_LABELS[line.category]] += _labor(line)
        top_labels, top_values = _ranked(counts)
        makes: dict[str, float] = defaultdict(float)

            makes[v.make] += 1

                _chart("revenue_by_category", "hbars", "Выручка по категориям работ", "rub",
                       *_split(by_category, "Выручка")),

    async def _clients(self, created, window: Window) -> tuple[int, int]:
        """Distinct clients with a visit created in `window`: new (first visit ever is in it) and repeat."""
        clients = {v.client_id for v in created}
        first = await self.repo.first_visit_at(list(clients))
        new = sum(1 for c in clients if first[c] in window)
        return new, len(clients) - new

    async def _extra_approved(self, created) -> int | None:
        """Share (%) of extra works the client approved, among visits created in the window; None if none."""
        extra = [line for line in await self.repo.work_lines([v.id for v in created]) if line.is_extra_work]
        return round(sum(1 for line in extra if line.approved_by_client) / len(extra) * 100) if extra else None
```

```ts
// web/src/ui/charts/format.ts
import type { StatsUnit } from '../../api/types';

const oneDecimal = new Intl.NumberFormat('ru-RU', { maximumFractionDigits: 1 });
const plain = (n: number) => oneDecimal.format(n).replace(/[  ]/g, ' ');

export function formatValue(value: number | string, unit: StatsUnit): string {
  if (typeof value === 'string') return value;
  switch (unit) {
    case 'rub': return `${plain(Math.round(value))} ₽`;
    case 'hours': return `${plain(value)} н/ч`;
    case 'days': return `${plain(value)} дн.`;
    case 'pct': return `${plain(value)} %`;
    default: return plain(value);
  }
}

/** Axis tick: roubles in thousands («15 т»), the rest as plain numbers. */
export function formatTick(value: number, unit: StatsUnit): string {
  return unit === 'rub' && value >= 1000 ? `${plain(value / 1000)} т` : plain(value);
}

/** Round axis top just above `v`, so ticks land on readable numbers. */
export function niceMax(v: number): number {
  if (v <= 0) return 1;
  const p = 10 ** Math.floor(Math.log10(v));
  for (const m of [1, 1.2, 1.5, 2, 2.5, 3, 4, 5, 6, 8, 10]) if (m * p >= v) return m * p;
  return 10 * p;
}
```

```tsx
// web/src/ui/charts/BarChart.tsx
import { useState } from 'react';
import type { StatsUnit } from '../../api/types';
import { formatTick, formatValue, niceMax } from './format';

export const W = 400;
export const H = 170;
const L = 38, R = 6, T = 14, B = 22;

/** Bar path with the data end (top) rounded and the baseline end square. */
export function roundedTop(x: number, top: number, w: number, h: number): string {
  const r = Math.min(4, h, w / 2);
  return `M${x},${top + h} V${top + r} Q${x},${top} ${x + r},${top} H${x + w - r} Q${x + w},${top} ${x + w},${top + r} V${top + h} Z`;
}

export interface Frame { y: (v: number) => number; ticks: number[]; step: number; barW: number; x: (i: number) => number; every: number; base: number }

/** Shared geometry for vertical bar charts over `n` steps with axis top `max`. */
export function frame(n: number, max: number): Frame {
  const step = (W - L - R) / n;
  const barW = Math.min(46, step * 0.6);
  return {
    y: (v) => H - B - (v / max) * (H - B - T),
    ticks: [0, 1, 2, 3].map((i) => (max * i) / 3),
    step, barW,
    x: (i) => L + step * i + (step - barW) / 2,
    every: Math.ceil(n / 6),
    base: H - B,
  };
}

export function Axis({ f, unit, labels }: { f: Frame; unit: StatsUnit; labels: string[] }) {
  return (
    <>
      {f.ticks.map((t) => (
        <g key={t}>
          <line x1={L} x2={W - R} y1={f.y(t)} y2={f.y(t)} className="chart-grid" />
          <text x={L - 6} y={f.y(t) + 4} textAnchor="end">{formatTick(t, unit)}</text>
        </g>
      ))}
      {labels.map((label, i) => i % f.every === 0 && (
        <text key={label} x={f.x(i) + f.barW / 2} y={H - 6} textAnchor="middle">{label}</text>
      ))}
    </>
  );
}

export function HitAreas({ f, n, onPick }: { f: Frame; n: number; onPick: (i: number) => void }) {
  return (
    <>
      {Array.from({ length: n }, (_, i) => (
        <rect key={i} data-testid={`bar-${i}`} className="chart-hit" x={L + f.step * i} y={T} width={f.step} height={f.base - T}
          onPointerEnter={() => onPick(i)} onClick={() => onPick(i)} />
      ))}
    </>
  );
}

export function BarChart({ title, labels, values, unit }: { title: string; labels: string[]; values: number[]; unit: StatsUnit }) {
  const [active, setActive] = useState(labels.length - 1);
  const f = frame(labels.length, niceMax(Math.max(...values)));
  return (
    <>
      <div className="chart-readout" data-testid="readout">{labels[active]} · <b>{formatValue(values[active], unit)}</b></div>
      <svg viewBox={`0 0 ${W} ${H}`} role="img" aria-label={title}>
        <Axis f={f} unit={unit} labels={labels} />
        {values.map((v, i) => v > 0 && (
          <path key={i} d={roundedTop(f.x(i), f.y(v), f.barW, f.base - f.y(v))} fill="var(--series-1)"
            opacity={i === active ? 1 : 0.75} />
        ))}
        <HitAreas f={f} n={labels.length} onPick={setActive} />
      </svg>
    </>
  );
}
```

```tsx
}) {

  const [bottom, top] = series;

        <Axis f={f} unit={unit} labels={labels} />
```

```python
# bot/handlers/stats.py
"""Statistics as a text summary: KPIs with change against the previous period, then each chart as █ bars."""
from aiogram.types import InlineKeyboardMarkup

from bot import nav
from bot.api_client import ApiClient
from bot.callback_ids import decode_id
from bot.formatting import format_number
from bot.handlers.navigation import master_dropdown

STATS_PERIODS = [("week", "Неделя"), ("month", "Месяц"), ("quarter", "Квартал"), ("year", "Год")]
STATS_SECTIONS = [("money", "Деньги"), ("flow", "Поток"), ("staff", "Сотрудники"), ("works", "Работы и клиенты")]
UNIT_SUFFIX = {"rub": " ₽", "hours": " н/ч", "days": " дн.", "pct": "%", "count": ""}
BAR_WIDTH = 8  # symbols for the largest value


def _number(value: float) -> str:
    return format_number(value) if float(value).is_integer() else f"{value:.1f}".replace(".", ",")


def format_value(value, unit: str) -> str:
    return str(value) if unit == "text" else _number(float(value)) + UNIT_SUFFIX[unit]


def _delta(kpi: dict) -> str:
    pct = kpi["delta_pct"]
    if not kpi["compare"] or pct is None:
        return ""
    return "  = 0%" if pct == 0 else f"  {'▲' if pct > 0 else '▼'} {abs(pct)}%"


def bar_lines(labels: list[str], values: list[float], unit: str) -> list[str]:
    top = max(values, default=0)
    if top <= 0:
        return ["Нет данных за период"]
    return [
        f"{label}  {'█' * max(1, round(v / top * BAR_WIDTH)) if v else '·'}  {format_value(v, unit)}"
        for label, v in zip(labels, values)
    ]


def _chart_values(chart: dict) -> list[float]:
    """A stacked chart reads as its totals in text."""
    return [sum(column) for column in zip(*(s["values"] for s in chart["series"]))]


@nav.screen("stats")
async def render_stats(api: ApiClient, user: dict, args: dict) -> nav.Rendered:
    period, opened = args.get("p", "month"), args.get(nav.OPEN, "-")
    period_label = dict(STATS_PERIODS).get(period, period)
    rows = nav.dropdown("Период", "p", STATS_PERIODS, period, opened)
    if user["role"] == "admin":
        section, master = args.get("sec", "money"), args.get("m", "-")
        data = await api.get_stats(section, period, master_id=None if master == "-" else decode_id(master))
        header = f"📊 Статистика · {period_label} · {dict(STATS_SECTIONS).get(section, section)}"
        rows += nav.dropdown("Раздел", "sec", STATS_SECTIONS, section, opened)
        rows += await master_dropdown(api, master, opened)
    else:
        data = await api.get_my_stats(period)
        header = f"📊 Моя статистика · {period_label}"
    lines = [header, ""]
    lines += [f"{k['label']}: {format_value(k['value'], k['unit'])}{_delta(k)}" for k in data["kpis"]]
    for chart in data["charts"]:
        lines += ["", f"{chart['title']}:", *bar_lines(chart["labels"], _chart_values(chart), chart["unit"])]
    return "\n".join(lines), InlineKeyboardMarkup(inline_keyboard=rows)
```
